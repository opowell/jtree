    /**
    * Called when a {@link Client} connected to a {@link Participant} starts this app.
    *
    * - client subscribes to this App's channel.
    * - client registers custom messages it may send.
    * - client registers automatic stage submission messages it may send.
    * - load custom behaviour [this.addClient]{@link App#addClient}.
    *
    * Called from:
    * - {@link App#participantBegin}
    *
    * @param  {Client} client The client who is connecting to this app.
    */
    addClientDefault(client) {
        client.socket.join(this.roomId());

        for (var i in this.messages) {
            var msg = this.messages[i];
            client.register(i, msg);
        }

        // Register for automatic stage messages.
        var app = this;
        var pageStages = this.pageStages();
        for (var s in pageStages) {
            var stageName = pageStages[s].name;

            // Listen to message from clients, once for each stage name (a client can start
            // more than one app, see Participant#exitRun); the functions below are this app's.
            if (client.stageListeners == null) {
                client.stageListeners = [];
            }
            if (!client.stageListeners.includes(stageName)) {
                client.stageListeners.push(stageName);
                client.on(stageName, function(data) { // stage messages are sent by default when submit button is clicked.
                    app.session.pushMessage(client, data.data, data.data.fnName);
                });
            }

            // Queue message.
            client[stageName] = function(data) {
                app.session.pushMessage(client, data, data.fnName + 'Process');
            }
            client[stageName].convertsOwnValues = true;

            // Process the message. Its values are converted by checkForm, not before (see Session#processMessage).
            client[stageName + 'Process'] = function(data) {

                app.jt.log('Server received auto-stage submission: ' + JSON.stringify(data));

                if (client.player() === null) {
                    return false;
                }

                if (client.player().roomId() !== data.playerRoomId || client.player().stage.id !== data.fnName) {
                    console.log('App.js, PLAYER ROOM ID DOES NOT MATCH, skipping submission: ' + client.player().stage.id + ' vs. ' + data.fnName + ', data=' + JSON.stringify(data));
                    return false;
                }

                var fields = Object.assign({}, data);
                delete fields.fnName;
                delete fields.playerRoomId;
                var errors = app.submitStage(client.player(), fields);
                if (errors !== null) {
                    client.socket.emit('formErrors', { stageId: data.fnName, errors: errors });
                    return false;
                }
                this.session.emitParticipantUpdates();

            };
            client[stageName + 'Process'].convertsOwnValues = true;
        }

        // Load custom code, overwrite default stage submission behavior.
        try {
            this.addClient(client);
        } catch(err) {
            console.log(err);
        }

    }

    /**
     * player submits the form of the stage they are playing: values by field name ('player.x',
     * 'group.x', 'participant.x', 'period.x' or 'app.x', or just 'x' for one of the stage's
     * formFields), as the page sends them. They are
     * checked and converted (see forms.js) and stored, and the stage ends for the player.
     * Once a player's time is up, what they submit is taken as it is: the values that pass, and
     * nothing for the others (as oTree does on a timeout).
     * @return {Object|null} If the form is refused, the messages by field name; else null.
     */
    submitStage(player, values) {
        // An input named as the field alone ('x', as oTree's pages write them): the one form
        // field of the stage that it names.
        var formFields = player.stage.formFields || [];
        for (var key of Object.keys(values)) {
            if (!key.includes('.')) {
                var matches = formFields.filter((f) => f.endsWith('.' + key));
                if (matches.length === 1 && !(matches[0] in values)) {
                    values[matches[0]] = values[key];
                    delete values[key];
                }
            }
        }
        var checked = checkForm(player.stage, player, values);
        if (checked.errors !== null && !player.timedOut) {
            return checked.errors;
        }
        var targets = {
            player: player,
            group: player.group,
            participant: player.participant,
            period: player.group.period,
            app: player.app().hostApp(),
        };
        for (var property in checked.values) {
            var dot = property.indexOf('.');
            var target = targets[property.substring(0, dot)];
            if (dot > 0 && target != null) {
                target[property.substring(dot + 1)] = checked.values[property];
            }
        }
        player.endStage(true);
        return null;
    }

    /**
     * Adds a stage, with contents loaded from a .jtt (or .js) file. In that file, `stage` is
     * the new stage and `app` is this app.
     * @param {string} name The id of the stage.
     * @param {string} [file] The stage's file, relative to this app's folder; by default
     * <name>.jtt (or .js).
     * @return {Stage} The new stage.
     */
    addStage(name, file) {
        var stage = this.newStage(name);
        var dir = path.resolve(path.dirname(this.appPath));
        var fn;
        if (file != null) {
            fn = path.resolve(dir, file);
        } else {
            fn = path.join(dir, name);
            if (fs.existsSync(fn + '.jtt')) {
                fn = fn + '.jtt';
            } else if (fs.existsSync(fn + '.js')) {
                fn = fn + '.js';
            }
        }
        stage.sourceFile = fn;
        runAppCode(Utils.readJS(fn), fn, { app: this, stage });
        return stage;
    }

    /**
     * Adds the app in file as a stage of this app: each time a group reaches the stage, its
     * players play that app, loaded afresh, with its own periods, groups and page, and then go
     * on to this app's next stage. Its players are those of the stage's period who play the
     * stage (see Stage#canPlayerParticipate). What the app's file sets of a stage's settings
     * (waitToStart, waitToEnd, waitForAllGroups, canPlayerParticipate, groupStart, playerStart,
     * playerEnd, groupEnd, allGroupsStart) is the stage's, unless this app sets them after.
     * A queue (.jtq) is an app whose stages are the apps it adds.
     * @param {string} file The app's file, or folder, relative to this app's folder.
     * @param {Object} [options] The app's option values, as for Session#addApp.
     * @param {string} [id] The id of the stage; by default, the app's. (Or addApp(file, id).)
     * @return {Stage} The new stage.
     */
    addApp(file, options, id) {
        if (typeof options === 'string') {
            id = options;
            options = undefined;
        }
        var appFile = path.resolve(this.queueDir || path.dirname(this.appPath), file);
        if (fs.existsSync(appFile) && fs.statSync(appFile).isDirectory()) {
            var main = ['app.jtt', 'app.js'].find((f) => fs.existsSync(path.join(appFile, f)));
            if (main == null) {
                throw new Error('addApp: no app.jtt in ' + appFile);
            }
            appFile = path.join(appFile, main);
        }
        var template = this.loadAppFile(appFile, options);
        if (id == null) {
            // The app's id; with a number after it, if that is taken (the same app again).
            id = template.shortId;
            for (let n = 2; this.stages.some((s) => s.id === id); n++) {
                id = template.shortId + '_' + n;
            }
        }
        var stage = this.newStage(id);
        stage.appFile = appFile;
        if (options != null && Object.keys(options).length > 0) {
            stage.appOptions = options;
        }
        stage.subApp = template;
        for (const k of APP_STAGE_SETTINGS) {
            if (Object.prototype.hasOwnProperty.call(template, k)) {
                stage[k] = template[k];
            }
        }
        return stage;
    }

    /**
     * A new App for the app in appFile (see addApp), in this app's session, with options set
     * before its code runs. For a queue, an app whose stages are the queue's apps.
     */
    loadAppFile(appFile, options) {
        var Queue = require('./Queue.js');
        var app = new App(this.session, this.jt, appFile);
        app.givenOptions = options;
        for (var name in options || {}) {
            app.setOptionValue(name, options[name]);
        }
        if (Queue.isQueueId(appFile)) {
            var queue = Queue.load(appFile, this.jt);
            if (queue.hasError) {
                throw new Error('Queue ' + appFile + ': ' + queue.errorMessage);
            }
            app.shortId = queue.shortId;
            app.title = queue.displayName;
            if (queue.description != null) {
                app.description = queue.description;
            }
            app.queueDir = queue.dir();
            // Its script adds its apps to the app, as it would to a session.
            queue.run(app);
            return app;
        }
        var dialect = dialectFor(appFile);
        if (dialect == null) {
            throw new Error('not an app: ' + appFile);
        }
        dialect.define(app);
        return app;
    }

    /**
     * The app that stage, a stage of this app given stages of its own (see Stage#newStage),
     * runs them as. It is this app with stage's stages, and periods and groups of stage's
     * (see Stage#runFor, which makes one for each period that reaches stage), so their code
     * and pages are this app's.
     * @param {Stage} stage
     * @return {App}
     */
    newBlock(stage) {
        var block = Object.create(this);
        block.stages = [];
        block.periods = [];
        block.parentStage = stage;
        block.duration = null;
        // What the stage sets itself (it has an app's defaults too, see Stage).
        var own = (name) => Object.prototype.hasOwnProperty.call(stage, name);
        var regroups = () => ['groupSize', 'numGroups', 'groupMatchingType'].some((k) => stage[k] != null) ||
            own('getGroupIdsForPeriod');
        // What the stage says, and is shown on pages as the app's.
        var fromStage = (name, get) => Object.defineProperty(block, name, {
            enumerable: true, configurable: true, get, set(v) { stage[name] = v; },
        });
        fromStage('numPeriods', () => stage.numPeriods != null ? stage.numPeriods : stage.repetitions);
        fromStage('groupSize', () => stage.groupSize);
        fromStage('numGroups', function() {
            if (regroups()) {
                return stage.numGroups;
            }
            return this.parentPeriod == null ? undefined : this.playingGroupIds().length;
        });
        fromStage('groupMatchingType', () => stage.groupMatchingType != null ? stage.groupMatchingType : 'STRANGER');
        fromStage('groupByArrival', () => stage.groupByArrival === true);
        // Its hooks are the stage's, not this app's.
        var method = (name, fn) => Object.defineProperty(block, name, { value: fn, writable: true, configurable: true });
        method('getGroupIdsForPeriod', function(period) {
            if (own('getGroupIdsForPeriod')) {
                return stage.getGroupIdsForPeriod(period);
            }
            if (regroups()) {
                return App.prototype.getGroupIdsForPeriod.call(this, period);
            }
            return this.playingGroupIds();
        });
        for (const hook of ['appStart', 'end', 'periodStart', 'periodEnd', 'participantStart', 'participantEnd']) {
            method(hook, function(...args) {
                if (own(hook)) {
                    stage[hook](...args);
                }
            });
        }
        return block;
    }

                    }

                    group.startStage(stage);
                }
            }
        }

    /**
     * TODO
     * Get group ids for their current period
     */
    getGroupIdsForPeriod(period) {
        var participants = this.periodParticipants(period);
        var numGroups = period.numGroups();
        var pIds = [];
        for (var p in participants) {
            pIds.push(p);
        }
        // Group IDs.
        var gIds = [];
        for (var g=0; g<numGroups; g++) {
            gIds.push([]);
        }

        for (let i=0; i<period.groups.length; i++) {
            let group = period.groups[i];
            for (let j=0; j<group.players.length; j++) {
                gIds[i].push(group.players[j].id);
                for (let k=0; k<pIds.length; k++) {
                    if (pIds[k] == group.players[j].id) {
                        pIds.splice(k, 1);
                    }
                }
            }
        }

        // Calculate number of elements per group
        var m = Math.floor((pIds.length-1) / numGroups) + 1;

        if (this.groupMatchingType === 'PARTNER_1122') {
            for (var g=period.groups.length; g<numGroups; g++) {
                for (var i=0; i<m; i++) {
                    gIds[g].push(pIds[0]);
                    pIds.splice(0, 1);
                }
            }
        } else if (this.groupMatchingType === 'PARTNER_1212') {
            for (var i=0; i<m; i++) {
                for (var g=period.groups.length; g<numGroups; g++) {
                    gIds[g].push(pIds[0]);
                    pIds.splice(0, 1);
                }
            }
        } else if (this.groupMatchingType === 'PARTNER_RANDOM') {
            if (period.id === 1) {
                period.getStrangerMatching(numGroups, pIds, gIds, m, period.groups.length);
            } else {
                var prevPeriod = period.prevPeriod();
                gIds = prevPeriod.groupIds();
            }
        } else if (this.groupMatchingType === 'STRANGER') {
            period.getStrangerMatching(numGroups, pIds, gIds, m, period.groups.length);
        }
        return gIds;
    }

    stripTag(tagName, text) {
        let strippedText = '';
        while (text.includes('<' + tagName)) {
            let start = text.indexOf('<' + tagName);
            if (start == -1) {
                break;
            }
            let nextStart = text.indexOf('<' + tagName, start + tagName.length + 1);
            let endTag = '/' + tagName + '>';
            let end = text.indexOf(endTag, start) + endTag.length;
            if (end == -1 || (nextStart > -1 && end > nextStart)) {
                endTag = '>';
                end = text.indexOf(endTag, start) + endTag.length;
            } 
            if (end == -1 + endTag.length) {
                break;
            }
            strippedText += text.substring(start, end);
            text = text.substring(0, start) + text.substring(end);
        }
        return [strippedText, text];
    }

            groupTableText.push('period.id' + this.outputDelimiter + 'group.id' + this.outputDelimiter + 'id' + this.outputDelimiter + tableHeaders.join(this.outputDelimiter));
            for (var i=0; i<this.periods.length; i++) {
                var period = this.periods[i];
                for (var j=0; j<period.groups.length; j++) {
                    var group = period.groups[j];
                    try {
                        var tabRows = group[table].rows;
                        for (var r=0; r<tabRows.length; r++) {
                            var row = tabRows[r];
                            var newLine = period.id + this.outputDelimiter + group.id + this.outputDelimiter + row.id + this.outputDelimiter;
                            newLine = this.appendValues(newLine, tableHeaders, row);
                            groupTableText.push(newLine);
                        }
                    } catch (err) {
                        
                    }
                }
            }
            fullText += groupTables[t].toUpperCase() + '\n';
            fullText += groupTableText.join('\n') + '\n';
        }

        if (playerHeaders.length > 0) {
            fullText += 'PLAYERS\n';
            fullText += playerText.join('\n') + '\n';
        }

        // Participants are with the app of the session.
        if (this.parentPeriod == null) {
            fullText += 'PARTICIPANTS\n';
            fullText += participantText.join('\n') + '\n';
        }

        return fullText;

    }

    setOptionValue(name, value) {
        this.optionValues[name] = value;
        var correctedValue = value;
        var isValid = false;
        let foundOpt = false;
        for (var opt in this.options) {
            var option = this.options[opt];
            if (option.name === name) {
                foundOpt = true;
                if (option.type === 'select') {
                    for (var i in option.values) {
                        if (option.values[i] == value) { // allow for coercion
                            correctedValue = option.values[i];
                            isValid = true;
                            break;
                        }
                    }
                } else if (option.type === 'number') {
                    correctedValue = value - 0; /** coerce to number*/
                    isValid = true;
                    break;
                } else if (option.type === 'text') {
                    isValid = true;
                    // no correction needed.
                }
            }
        }
        if (isValid || !foundOpt) {
            this[name] = correctedValue;
        }
    }

    /**
     * Called when a participant begins this app.
     * - For each of the [Clients]{@link Client} of this participant, call {@link App.addClientDefault}.
     * - Set the participant's periodIndex to -1.
     * - Participant notifies clients about appIndex.
     * - Move participant to next period {@link App.participantMoveToNextPeriod}.
     * - Participant notified clients about starting new app.
     *
     * @param  {Participant} participant The participant.
     */
    participantBegin(participant) {

        // The stages of an app within a stage given stages of its own (see Stage#newStage)
        // are on the page of the app it is in.
        var newPage = this.hostApp() === this;
        if (newPage) {
            for (var c in participant.clients) {
                var client = participant.clients[c];
                this.addClientDefault(client);
            }
        }

        participant.periodIndex = -1;
        if (this.parentPeriod == null) {
            participant.emit('participantSetAppIndex', {appIndex: this.indexInSession()});
        }

        let duration = this.getParticipantDuration(participant);
        if (duration != null) {
            if (this.parentPeriod != null) {
                // An app within a stage (see Participant#enterRun): they leave it when time is up.
                var run = this;
                participant.frames[participant.frames.length - 1].timer = new Timer.new(
                    function() {
                        participant.session.addMessageToStartOfQueue(participant, run, 'leaveRun');
                    },
                    duration*1000
                );
            } else {
                participant.appTimer = new Timer.new(
                    function() {
                        participant.session.addMessageToStartOfQueue(participant, {}, 'endCurrentApp');
                    },
                    duration*1000
                );
            }
        }
        this.participantStart(participant);
        this.participantMoveToNextPeriod(participant);

        if (newPage) {
            participant.emit('start-new-app'); /** refresh clients.*/
        }
    }

     * FUNCTIONALITY
     * - If participant is currently in a period, end it ({@link Period.participantEnd}).
     * - Increment participant's period index.
     * - Save participant ({@link Participant.save}).
     * - If participant is has finished all periods in this app, move to next app ({@link Session.participantMoveToNextApp}).
     * - Otherwise, begin new period ({@link App.participantBeginPeriod}).

     * CALLED FROM
     * - {@link App#playerMoveToNextStage}
     * - {@link App#participantBegin}
     *
     * @param  {type} participant description
     * @return {type}             description
     */
     participantMoveToNextPeriod(participant) {
         // If in the last period of app, move to next app.
         if (participant.periodIndex >= this.numPeriods - 1) {
             this.session.participantMoveToNextApp(participant);
         }

         // Move to the next period of this app.
         else {
             // If not in the first period, end the previous period for this participant.
             if (participant.periodIndex > -1) {
                 participant.player.period().participantEnd(participant);
             }

             // Move to next period.
             participant.periodIndex++;
             participant.save();
             this.participantBeginPeriod(participant);
         }
     }

    appendValues(newLine, headers, obj) {
        for (var i=0; i<headers.length; i++) {
            var header = headers[i];
            var value = obj[header];
            if (value !== undefined && value !== null) {

                // If value is an object, change it to string.
                if (typeof value === 'object') {
                    value = JSON.stringify(obj[header]);
                }

                // If value is a string, replace any commas.
                if (typeof value === 'string') {
                    value = value.replace(/,/g, '--');
                }

                // Append the value.
                newLine += value;
            }

            if (i<headers.length-1) {
                newLine += this.outputDelimiter;
            }
        }
        return newLine;
    }
