		startwaitforall = TRUE;

		singleentry = FALSE;
		singleentrycontinuation = FALSE;
		timeouttype = always;
		timeout = 45;
		program{
			table = subjects;
			do{
				Value = rounddown(random() * (MaxValue + 1), 1);
			}
		}
		program{
			table = globals;
			do{
				Price = 100;
							PriceStep = 5;
							// The player who bought; 0 while nobody has.
							Buyer = 0;
							SalePrice = 0;
							// The price falls every 2 seconds, until someone buys or it is 0.
							later(2) repeat {
								if (Buyer == 0 & Price > 0) {
									Price = Price - PriceStep;
								}
							}
			}
		}

			usesbg = TRUE;
			withalertscreen = FALSE;

						message = "Another bidder was quicker.";
						yesButtonStr = "";
						noButtonStr = "OK";
					}
					program{
						table = subjects;
						do{
							Won = 1;
										\Buyer = Player;
										\SalePrice = Price;
										// The auction is over for everyone in the group.
										subjects.do {
											if (Group == :Group) {
												LeaveStage = 1;
											}
										}
						}
					}
				}
			}
		}

			noalertscreen = FALSE;
