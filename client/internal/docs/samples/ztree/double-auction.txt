		program{
			table = subjects;
			do{
				// Unit k is worth Values[k] to a buyer, and costs a seller Costs[k].
						array Values[4];
						Values[1] = 400;
						Values[2] = 300;
						Values[3] = 200;
						Values[4] = 200;
						array Costs[4];
						Costs[1] = 20;
						Costs[2] = 120;
						Costs[3] = 220;
						Costs[4] = 220;
						MaxUnits = 4;
						MaxPrice = 1000;
						UnitsTraded = 0;
			}
		}
		program{
			table = contracts;
			do{
				// An offer: a bid has Buyer (its maker) and Seller = -1; an ask the other way round.
						// A trade has both; a withdrawn offer has -2 for the side it lacked.
						Buyer = 0;
						Seller = 0;
						TradePrice = 0;
			}
		}

		singleentrycontinuation = FALSE;

		timeout = TradingTime;

						IsBuyer = if(Player <= 4, 1, 0);

						message = "You have traded all your units.";
						yesButtonStr = "";
						noButtonStr = "OK";
					}
					checker{
						condition = contracts.count(Seller == -1 & Buyer > 0) == 0 | Price > contracts.maximum(Seller == -1 & Buyer > 0, Price);
						message = "Your bid must be higher than the best bid.";
						yesButtonStr = "";
						noButtonStr = "OK";
					}
					program{
						table = contracts;
						do{
							Buyer = :Subject;
							Seller = -1;

				allemptyallowed = FALSE;

				}

						message = "You have traded all your units.";

						noButtonStr = "OK";
					}
					program{
						table = contracts;
						do{
							Buyer = :Subject;
									TradePrice = Price;
									subjects.do {
										if (Subject == :Buyer) {
											UnitsTraded = UnitsTraded + 1;
											Profit = Profit + Values[UnitsTraded] - :Price;
										}
										if (Subject == :Seller) {
											UnitsTraded = UnitsTraded + 1;
											Profit = Profit + :Price - Costs[UnitsTraded];
										}
									}
									// The traders' other offers are withdrawn.
									contracts.do {
										if (Seller == :Seller & Buyer == -1) {
											Buyer = -2;
										}
										if (Buyer == :Buyer & Seller == -1) {
											Seller = -2;
										}
									}
						}
					}
				}
			}
			selectcontractbox "Bids"{
				hasframe = TRUE;
				width = 30%;
				table = contracts;
				buttonposition = BOTTOMRIGHT;
				buttonsequence = HORIZONTAL;
				emptyallowed = FALSE;
				allemptyallowed = FALSE;
				condition = Seller == -1 & Buyer > 0;
				sortfields = "-Price";
				tablelayout = LIST;
				scrollatstart = FALSE;
				scrollatEnd = FALSE;
				selectbest = FALSE;

					label = "Bid";
					variable = Price;
					format = "1";
					input = FALSE;
